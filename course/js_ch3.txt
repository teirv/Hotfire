=U regex | Regular Expressions | Book section 3.1. Patterns for matching text.
=L regex1 | Matching patterns
@T
A **regular expression** is a pattern for text, written between slashes. `.test()` asks whether a string matches.

`\d` is a digit, `\w` is a letter, digit or underscore, `\s` is a space, and `.` is any character.

`+` means one or more, `*` means zero or more, and `?` means optional. `^` is the start, and `$` is the end.
~code
console.log(/\d+/.test("Tank 7"));
console.log(/^\d+$/.test("Tank 7"));
console.log(/^\d+$/.test("707"));
console.log(/colou?r/.test("color"));
@T
Square brackets are a choice of characters: `[a-z]`. Flags go after the last slash: `i` ignores case, and `g` finds all matches.

`match` finds matches. `replace` swaps them. Parentheses **capture** a piece for later.
~code
console.log("a1 b22".match(/\d+/g));
console.log("Hello".replace(/l/g, "L"));

let m = "2026-11-03".match(/(\d+)-(\d+)-(\d+)/);
console.log(m[1], m[2]);
@P
What does this print?
~code
console.log(/^[A-Z]\d$/.test("B7"), /^[A-Z]\d$/.test("B77"));
~opts
* true false
- true true
- false false
~why
The pattern is exactly one capital letter, then exactly one digit.
@Q
Which regex matches a 5-digit ZIP code and nothing more?
~opts
* `/^\d{5}$/`
- `/\d{5}/`
- `/^d5$/`
~why
`{5}` means exactly five. Without `^` and `$`, longer strings would also match.

=L class1 | Writing a class
@T
A **class** is a blueprint for making objects. `constructor` runs when you use `new`. `this` is the object being built. Methods go inside without the word `function`.
~code
class Sensor {
  constructor(name, limit) {
    this.name = name;
    this.limit = limit;
  }

  isHigh(value) {
    return value > this.limit;
  }
}

let pt = new Sensor("PT-101", 500);
console.log(pt.name);
console.log(pt.isHigh(600));
@T
`extends` makes a class that inherits from another. `super()` calls the parent's constructor. A child can **override** a method.
~code
class Tool {
  constructor(name) {
    this.name = name;
  }
  describe() {
    return `Tool: ${this.name}`;
  }
}

class Wrench extends Tool {
  constructor(size) {
    super("wrench");
    this.size = size;
  }
  describe() {
    return `${super.describe()}, ${this.size} mm`;
  }
}

console.log(new Wrench(10).describe());
@T

~code
class Account {

  static count = 0;

  constructor() {
    Account.count++;
  }
  deposit(n) {
    this.#balance += n;
  }
  get balance() {
    return this.#balance;
  }
}

let a = new Account();
a.deposit(50);
console.log(a.balance, Account.count);

=L mod1 | Modules
@T
A **module** is a file with its own scope. It shares things with `export`, and other files bring them in with `import`.

```
// math.js
export function add(a, b) { return a + b; }
export default function square(n) { return n * n; }

// app.js
import square, { add } from "./math.js";
```

In HTML, load it with `<script type="module" src="app.js"></script>`. Modules run deferred and are strict automatically. The app runs one file at a time, so this lesson isn't run here.

- None
~why

@T

ctx.moveTo(110, 20);

ctx.arc(60, 90, 20, 0, 2 * Math.PI);

ctx.fill();
@T
**Transformations** move the drawing grid. `translate` shifts it, `rotate` turns it (in radians), and `scale` resizes it. `save` and `restore` undo them.

For **animation**, clear the canvas, redraw, and use `requestAnimationFrame` to repeat.
~html
<canvas id="c" width="240" height="120"></canvas>
~code
let ctx = document.getElementById("c").getContext("2d");
let x = 0;

function frame() {
  ctx.clearRect(0, 0, 240, 120);
  ctx.save();
  ctx.translate(x, 60);
  ctx.rotate(x / 30);
  ctx.fillStyle = "firebrick";
  ctx.fillRect(-15, -15, 30, 30);
  ctx.restore();
  x = (x + 2) % 240;
  requestAnimationFrame(frame);
}

frame();

=U promises | Promises and async | Book sections 3.12 and 3.13. Working with waiting.
=L prom1 | Promises
@T
A **promise** stands for a value that isn't ready yet. It's like a claim ticket at a repair shop. It ends up **fulfilled** with a value or **rejected** with an error.

`.then` runs on success, and `.catch` runs on failure. `.finally` always runs.
~code
function getPart(ok) {
  return new Promise((resolve, reject) => {
    setTimeout(() => {
      if (ok) resolve("valve");
      else reject(new Error("Out of stock"));
    }, 20);
  });
}

getPart(true)
  .then((part) => console.log("Got", part))
  .catch((e) => console.log(e.message));

getPart(false)
  .then((part) => console.log("Got", part))
  .catch((e) => console.log(e.message));
@T
Each `.then` hands its return value to the next, forming a chain. `Promise.all` waits for several promises at once.
~code
Promise.resolve(2)
  .then((n) => n * 3)
  .then((n) => console.log(n));

Promise.all([Promise.resolve(1), Promise.resolve(2)])
  .then((values) => console.log(values));

~code

console.log("B");
~opts
* B
A
- A
B
- A
~why
`.then` always waits until the current code has finished.

Inside `new Promise((resolve, reject) => ...)`, what does calling `reject` do?
~opts
* Marks the promise as failed, so `.catch` runs
- Restarts the promise
- Ends the whole program
~why
`resolve` is the success path.

function drawFrog(ctx, x, y, angle) {
  ctx.save();
  ctx.translate(x, y);
  ctx.rotate(angle);
  ctx.fillRect(-10, -10, 20, 20);
  ctx.restore();
}

let ctx = document.getElementById("c").getContext("2d");

Write `circleArea(r)` that returns a promise. It resolves with the area rounded to 2 decimals as a string (use `toFixed(2)`). If `r` is not positive, reject with a new Error `Radius must be positive`.

~sol
function circleArea(r) {
  return new Promise((resolve, reject) => {
    if (r <= 0) {
      reject(new Error("Radius must be positive"));
    } else {
      resolve((Math.PI * r * r).toFixed(2));
    }
  });
}
~hint
Inside new Promise, call reject for bad input and resolve otherwise.
~after
await circleArea(1).then(console.log);
await circleArea(-1).catch((e) => console.log(e.message));
